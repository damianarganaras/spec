-- table memory_nodes
CREATE TABLE memory_nodes (
    id TEXT PRIMARY KEY,
    memory_key TEXT NOT NULL,
    type TEXT NOT NULL CHECK (type IN ('rule', 'decision')),
    scope TEXT NOT NULL DEFAULT 'project',
    status TEXT NOT NULL DEFAULT 'active' CHECK (status IN ('active', 'superseded', 'deleted')),
    content TEXT NOT NULL,
    justification TEXT NOT NULL DEFAULT '',
    superseded_by TEXT REFERENCES memory_nodes(id) DEFERRABLE INITIALLY DEFERRED,
    source TEXT NOT NULL,
    confidence REAL NOT NULL,
    created_at TEXT NOT NULL
  , superseded_at TEXT)

-- index idx_memory_nodes_key_active
CREATE UNIQUE INDEX idx_memory_nodes_key_active
    ON memory_nodes(memory_key) WHERE status = 'active'

-- index idx_memory_nodes_scope_type
CREATE INDEX idx_memory_nodes_scope_type
    ON memory_nodes(scope, type, status)

-- table memory_fts
CREATE VIRTUAL TABLE memory_fts USING fts5(
    content,
    content = 'memory_nodes',
    content_rowid = 'rowid',
    tokenize = 'unicode61 remove_diacritics 1'
  )

-- table memory_fts_data
CREATE TABLE 'memory_fts_data'(id INTEGER PRIMARY KEY, block BLOB)

-- table memory_fts_idx
CREATE TABLE 'memory_fts_idx'(segid, term, pgno, PRIMARY KEY(segid, term)) WITHOUT ROWID

-- table memory_fts_docsize
CREATE TABLE 'memory_fts_docsize'(id INTEGER PRIMARY KEY, sz BLOB)

-- table memory_fts_config
CREATE TABLE 'memory_fts_config'(k PRIMARY KEY, v) WITHOUT ROWID

-- trigger memory_fts_ai
CREATE TRIGGER memory_fts_ai AFTER INSERT ON memory_nodes BEGIN
    INSERT INTO memory_fts(rowid, content) VALUES (new.rowid, new.content);
  END

-- trigger memory_fts_ad
CREATE TRIGGER memory_fts_ad AFTER DELETE ON memory_nodes BEGIN
    INSERT INTO memory_fts(memory_fts, rowid, content) VALUES ('delete', old.rowid, old.content);
  END

-- trigger memory_fts_au
CREATE TRIGGER memory_fts_au AFTER UPDATE ON memory_nodes BEGIN
    INSERT INTO memory_fts(memory_fts, rowid, content) VALUES ('delete', old.rowid, old.content);
    INSERT INTO memory_fts(rowid, content) VALUES (new.rowid, new.content);
  END
